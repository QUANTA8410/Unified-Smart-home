  /// Automatically checks clipboard, scans local LAN or auto-scans Realme/Philips cameras the instant the user returns to the app
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    final vt = widget.brand.vendorType;
    if (state == AppLifecycleState.resumed && !_isLinking) {
      if (_waitingForRealmeReturn && (vt == 'realme_camera' || vt == 'realme_cam')) {
        _waitingForRealmeReturn = false;
        _scanLocalSubnetForRealmeDevices();
      } else if (_waitingForPhilipsReturn && vt == 'philips_camera') {
        _waitingForPhilipsReturn = false;
        _scanLocalSubnetForPhilipsDevices();
      } else if (vt == 'wipro_tuya') {
        _scanAndLinkWiproBulbFromLan();
      } else if (vt != 'philips_hue' &&
          vt != 'philips_wiz' &&
          vt != 'realme_camera' &&
          vt != 'realme_cam' &&
          vt != 'philips_camera') {
        _checkClipboardForTokenAutomatically();
      }
    }
  }

  Future<void> _checkClipboardForTokenAutomatically() async {
    try {
      final clip = await Clipboard.getData(Clipboard.kTextPlain);
      final text = clip?.text?.trim() ?? '';
      // Match SmartThings UUID (36 chars) or LG ThinQ PAT / Home Assistant JWT (>= 32 chars, no spaces)
      final looksLikeToken = text.length >= 32 &&
          !text.contains(' ') &&
          !text.startsWith('http');

      if (looksLikeToken && mounted) {
        _manualTokenCtrl.text = text;
        setState(() {
          _statusMessage = 'Token detected from clipboard! Linking ${widget.brand.brandName}...';
        });
        await _completeAccountLinking(text);
      }
    } catch (_) {}
  }

  /// Opens the official login page in Android's native Chrome Custom Tab:
  /// - 10x faster rendering (no WebView GPU bottleneck)
  /// - Full "Sign in with Google" / Apple / Samsung Pass support
  /// - Full 3rd-party cookie & popup window support
  Future<void> _launchNativeChromeCustomTab() async {
    final uri = Uri.parse(widget.brand.loginUrl);
    try {
      final launched = await launchUrl(
        uri,
        mode: LaunchMode.inAppBrowserView,
        browserConfiguration: const BrowserConfiguration(showTitle: true),
      );
      if (!launched) {
        await launchUrl(uri, mode: LaunchMode.externalApplication);
      }
    } catch (e) {
      // If Custom Tabs fail on a custom ROM, initialize the lightweight embedded WebView
      _initLightweightEmbeddedWebView();
    }
  }

  void _initLightweightEmbeddedWebView() {
    final ctrl = WebViewController()
      ..setJavaScriptMode(JavaScriptMode.unrestricted)
      ..setBackgroundColor(const Color(0xFF0A0F1D))
      ..setNavigationDelegate(
        NavigationDelegate(
          onProgress: (p) {
            if (mounted) setState(() => _loadProgress = p / 100.0);
          },
          onNavigationRequest: (req) {
            if (req.url.startsWith('unifiedhome://')) {
              final uri = Uri.parse(req.url);
              final token = uri.queryParameters['access_token'] ?? uri.queryParameters['code'];
              if (token != null && token.isNotEmpty) {
                _completeAccountLinking(token);
              }
              return NavigationDecision.prevent;
            }
            return NavigationDecision.navigate;
          },
        ),
      )
      ..loadRequest(Uri.parse(widget.brand.loginUrl));

    setState(() {
      _webController = ctrl;
      _useEmbeddedFallback = true;
    });
  }

      final bulb = RealSmartDevice(
        id: devId,
        name: name,
        room: 'Living Room',

        protocol: ConnectionProtocol.tuyaLocalLan,
        credentials: {
          'ip': ip,
          'localKey': key,
          'devId': devId,

        isOnline: true,

  Future<void> _completeAccountLinking(String token) async {
    final cleanToken = token.trim();
    if (cleanToken.isEmpty || _isLinking) return;

    setState(() {
      _isLinking = true;
      _statusMessage = 'Verifying ${widget.brand.brandName} credentials with cloud...';
    });
    HapticFeedback.heavyImpact();

    final home = context.read<RealHomeProvider>();
    try {
      int importedCount = 0;
      if (widget.brand.vendorType == 'lg_thinq') {
        final lgDevices = await home.zeroTouch.discoverLgThinqDevices(patToken: cleanToken);
        for (final d in lgDevices) {
          await home.addDirectLanDevice(d);
        }
        importedCount = lgDevices.length;
      } else if (widget.brand.vendorType == 'smartthings') {
        importedCount = await home.linkSmartThingsAccount(cleanToken);
      } else if (widget.brand.vendorType == 'home_assistant') {
        importedCount = await home.linkHomeAssistantHub(
          'http://homeassistant.local:8123',
          cleanToken,
        );
      } else {
        importedCount = await home.linkSmartThingsAccount(cleanToken);
      }

      if (mounted) {
        Navigator.of(context).pop(true);
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            backgroundColor: const Color(0xFF30D158),
            content: Text(
              'Connected ${widget.brand.brandName}! Imported $importedCount real device(s).',
              style: const TextStyle(color: Colors.white, fontWeight: FontWeight.bold),
            ),
          ),
        );
      }
    } catch (e) {
      if (mounted) {
        setState(() {
          _isLinking = false;
          _statusMessage = 'Connection Error: ${e.toString().replaceAll("Exception: ", "")}';
        });
      }
    }
  }

  Future<void> _scanLocalSubnetForRealmeDevices() async {
    if (_isScanningCamera) return;
    setState(() {
      _isScanningCamera = true;
      _statusMessage = 'Scanning local Wi-Fi subnet for Realme Cam (Ports 554, 8554, 8000)...';
    });

    final discovered = <String>{};
    try {
      String subnetPrefix = '192.168.1.';
      try {
        final interfaces = await NetworkInterface.list(
          type: InternetAddressType.IPv4,
          includeLoopback: false,
        );
        for (final iface in interfaces) {
          for (final addr in iface.addresses) {
            if (!addr.isLoopback && addr.address.contains('.')) {
              final parts = addr.address.split('.');
              if (parts.length == 4) {
                subnetPrefix = '${parts[0]}.${parts[1]}.${parts[2]}.';
                break;
              }
            }
          }
        }
      } catch (_) {}

      final portsToProbe = [554, 8554, 8000];
      const batchSize = 25;
      for (int i = 1; i <= 254; i += batchSize) {
        final batch = <Future<void>>[];
        for (int j = i; j < i + batchSize && j <= 254; j++) {
          final host = '$subnetPrefix$j';
          batch.add(() async {
            for (final port in portsToProbe) {
              try {
                final s = await Socket.connect(host, port, timeout: const Duration(milliseconds: 350));
                try {
                  discovered.add(host);
                  break;
                } finally {
                  s.destroy();
                }
              } catch (_) {}
            }
          }());
        }
        await Future.wait(batch);
        if (discovered.isNotEmpty && mounted) {
          setState(() {
            _discoveredRealmeIps = discovered.toList();
            if (_camIpCtrl.text.isEmpty) {
              _camIpCtrl.text = discovered.first;
            }
          });
        }
      }
    } catch (_) {}

    if (mounted) {
      setState(() {
        _isScanningCamera = false;
        _discoveredRealmeIps = discovered.toList();
        if (discovered.isNotEmpty) {
          if (_camIpCtrl.text.isEmpty || !_discoveredRealmeIps.contains(_camIpCtrl.text)) {
            _camIpCtrl.text = discovered.first;
          }
          _statusMessage = 'Found ${discovered.length} camera target(s): ${discovered.join(", ")}';
        } else {
          _statusMessage = 'No camera auto-detected on ports 554/8554/8000. Enter Camera IP manually.';
        }
      });
    }
  }

          _statusMessage = 'Found ${discovered.length} Philips camera target(s): ${discovered.join(", ")}';
        } else {
          _statusMessage = 'No Philips camera auto-detected on ports 554/8554/80/8080. Enter Camera IP manually.';
        }
      });
    }
  }

  Future<void> _connectCamera() async {
    final ip = _camIpCtrl.text.trim();
    final port = _camPortCtrl.text.trim();
    final username = _camUserCtrl.text.trim();
    final password = _camPassCtrl.text;
    final customName = _camNameCtrl.text.trim();
    final room = _camRoomCtrl.text.trim().isEmpty ? 'Living Room' : _camRoomCtrl.text.trim();

    if (ip.isEmpty) {
      setState(() => _statusMessage = 'Please enter the Camera IP address.');
      return;
    }

    final isRealme = widget.brand.vendorType == 'realme_camera' || widget.brand.vendorType == 'realme_cam';
    final brand = isRealme ? 'realme' : 'philips';

    setState(() {
      _isLinking = true;
      _statusMessage = 'Probing RTSP socket connection at $ip:${port.isEmpty ? "554" : port}...';
    });
    HapticFeedback.heavyImpact();

    final home = context.read<RealHomeProvider>();
    final messenger = ScaffoldMessenger.of(context);
    final rootContext = Navigator.of(context).context;

    try {
      final cam = await home.linkRtspCamera(
        brand: brand,
        ip: ip,
        port: port.isEmpty ? '554' : port,
        username: username,
        password: password,
        room: room,
        customName: customName.isNotEmpty ? customName : null,
      );

      if (mounted) {
        Navigator.of(context).pop(true);
        messenger.showSnackBar(
          SnackBar(
            backgroundColor: const Color(0xFF30D158),
            content: Text(
              'Connected ${cam.name}! Live 1080p RTSP Stream & Controls Active.',
              style: const TextStyle(color: Colors.white, fontWeight: FontWeight.bold),
            ),
          ),
        );

        if (rootContext.mounted) {
          showModalBottomSheet(
            context: rootContext,
            isScrollControlled: true,
            backgroundColor: Colors.transparent,
            barrierColor: Colors.black.withValues(alpha: 0.65),
            builder: (ctx) => CameraCockpitSheet(
              device: cam,
              iot: home.iot,
              onStateChanged: () => home.notifyUi(),
            ),
          );
        }
      }
    } catch (e) {
      if (mounted) {
        setState(() {
          _isLinking = false;
          _statusMessage = 'Camera Connection Error: ${e.toString().replaceAll("Exception: ", "")}';
        });
      }
    }
  }

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _scanController = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 4),
    )..repeat(reverse: true);

    _initRtsPlayer();
  }

  void _initRtsPlayer() {
    try {
      final player = Player(
        configuration: const PlayerConfiguration(
          bufferSize: 2 * 1024 * 1024,
          protocolWhitelist: [
            'rtsp',
            'rtp',
            'tcp',
            'udp',
            'tls',
            'data',
            'file',
            'http',
            'https',
            'crypto',
          ],
        ),
      );

      final controller = VideoController(
        player,
        configuration: const VideoControllerConfiguration(
          enableHardwareAcceleration: true,
        ),
      );

      _player = player;
      _videoController = controller;

      _playingSub = player.stream.playing.listen((playing) {
        if (mounted) setState(() => _isPlaying = playing);
      });

      _bufferingSub = player.stream.buffering.listen((buffering) {
        if (mounted) setState(() => _isBuffering = buffering);
      });

      _errorSub = player.stream.error.listen((error) {
        if (mounted && error.isNotEmpty) {
          setState(() => _playerError = error);
        }
      });

      _videoParamsSub = player.stream.videoParams.listen((params) {
        if (mounted && params.w != null && params.h != null) {
          setState(() {
            _streamTelemetry = '${params.w}x${params.h} • 30 FPS • H.264';
          });
        }
      });

      _startStreamPlayback();
    } catch (e) {
      debugPrint('RTSP player initialization note: $e');
    }
  }

  String _getActiveStreamUrl() {
    final rtspUrl = widget.device.credentials['rtspUrl'] ?? '';
    if (rtspUrl.isEmpty) return '';

    if (_useSubStream) {
      final sub = widget.device.credentials['subStreamUrl'] ??
          widget.device.credentials['rtspSubUrl'];
      if (sub != null && sub.toString().isNotEmpty) {
        return sub.toString();
      }
      if (rtspUrl.contains('/stream1')) {
        return rtspUrl.replaceAll('/stream1', '/stream2');
      } else if (rtspUrl.contains('/ch0')) {
        return rtspUrl.replaceAll('/ch0', '/ch1');
      } else if (rtspUrl.contains('/live/0')) {
        return rtspUrl.replaceAll('/live/0', '/live/1');
      } else if (rtspUrl.contains('/onvif1')) {
        return rtspUrl.replaceAll('/onvif1', '/onvif2');
      } else if (rtspUrl.contains('/h264Preview_01_main')) {
        return rtspUrl.replaceAll('/h264Preview_01_main', '/h264Preview_01_sub');
      }
    }
    return rtspUrl;
  }

  Future<void> _startStreamPlayback() async {
    final isPrivacy = !widget.device.isOn;
    if (_player == null || isPrivacy || !widget.device.isOnline) {
      try {
        await _player?.stop();
      } catch (_) {}
      return;
    }

    final url = _getActiveStreamUrl();
    if (url.isEmpty) return;

    try {
      await _player?.setVolume(_isMuted ? 0.0 : 100.0);
      await _player?.open(Media(url));
    } catch (e) {
      if (mounted) setState(() => _playerError = '$e');
    }
  }

  Future<void> _toggleStreamProfile() async {
    HapticFeedback.lightImpact();
    setState(() {
      _useSubStream = !_useSubStream;
      _playerError = null;
      _streamTelemetry = _useSubStream
          ? '640x360 • 15 FPS • 512 Kbps'
          : '1080P • 30 FPS • 2.4 Mbps • H.264';
    });
    await _startStreamPlayback();
  }

  Future<void> _toggleAudioMute() async {
    HapticFeedback.selectionClick();
    setState(() {
      _isMuted = !_isMuted;
    });
    try {
      await _player?.setVolume(_isMuted ? 0.0 : 100.0);
    } catch (_) {}
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (_isDisposed || _player == null) return;
    if (state == AppLifecycleState.paused || state == AppLifecycleState.inactive) {
      try {
        _player?.pause();
      } catch (_) {}
    } else if (state == AppLifecycleState.resumed && mounted && widget.device.isOn) {
      try {
        _player?.play();
      } catch (_) {}
    }
  }

  @override
  void dispose() {
    if (_isDisposed) return;
    _isDisposed = true;
    WidgetsBinding.instance.removeObserver(this);
    _scanController.dispose();
    _playingSub?.cancel();
    _bufferingSub?.cancel();
    _errorSub?.cancel();
    _videoParamsSub?.cancel();
    try {
      _player?.stop();
    } catch (_) {}
    try {
      _player?.dispose();
    } catch (_) {}
    super.dispose();
  }

  Future<void> _handlePtz(String direction) async {
    HapticFeedback.mediumImpact();
    setState(() => _ptzStatus = 'Moving ${direction.toUpperCase()}...');
    try {
      await widget.iot.sendCameraPtzCommand(widget.device, direction);
      widget.onStateChanged();
      if (mounted) {
        setState(() {
          final pan = (widget.device.liveAttributes['panAngle'] as num?)?.toDouble() ?? 0.0;
          final tilt = (widget.device.liveAttributes['tiltAngle'] as num?)?.toDouble() ?? 0.0;
          _ptzStatus = 'Pan: ${pan.round()}° • Tilt: ${tilt.round()}°';
        });
      }
    } catch (e) {
      if (mounted) {
        setState(() => _ptzStatus = 'PTZ Error: $e');
      }
    }
  }

  Future<void> _launchStream() async {
    final rtspUrl = _getActiveStreamUrl();
    if (rtspUrl.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('No RTSP stream URL configured for this camera.')),
      );
      return;
    }

    HapticFeedback.selectionClick();
    final uri = Uri.parse(rtspUrl);
    try {
      final launched = await launchUrl(uri, mode: LaunchMode.externalApplication);
      if (!launched) {
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('Could not launch external player for $rtspUrl')),
          );
        }
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Stream player error: $e')),
        );
      }
    }
  }

  Future<void> _toggleHardwareFeature(String featureKey) async {
    HapticFeedback.lightImpact();
    final attrs = Map<String, dynamic>.from(widget.device.liveAttributes);

    if (featureKey == 'privacyMask') {
      final currentPrivacy = !widget.device.isOn;
      final targetPrivacy = !currentPrivacy;
      widget.device.isOn = !targetPrivacy;
      attrs['privacyMode'] = targetPrivacy;
      widget.device.liveAttributes = attrs;
      widget.iot.computeDeviceLiveWattage(widget.device);
      try {
        await widget.iot.setPower(widget.device, !targetPrivacy);
      } catch (_) {}
      await _startStreamPlayback();
    } else {
      final current = attrs[featureKey] == true;
      attrs[featureKey] = !current;
      widget.device.liveAttributes = attrs;
    }

    widget.onStateChanged();
    setState(() {});
  }

class _CockpitReticlePainter extends CustomPainter {
  final double scanProgress;
  final bool isPrivacy;
  final bool hasLiveVideo;

  const _CockpitReticlePainter({
    required this.scanProgress,
    required this.isPrivacy,
    this.hasLiveVideo = false,
  });

  @override
  void paint(Canvas canvas, Size size) {
    if (isPrivacy) return;

    final bracketPaint = Paint()
      ..color = Colors.white.withValues(alpha: hasLiveVideo ? 0.4 : 0.18)
      ..strokeWidth = 1.2
      ..style = PaintingStyle.stroke;

    // Viewfinder Corner Brackets
    const corner = 18.0;
    const pad = 14.0;

    // Top-Left
    canvas.drawLine(const Offset(pad, pad), const Offset(pad + corner, pad), bracketPaint);
    canvas.drawLine(const Offset(pad, pad), const Offset(pad, pad + corner), bracketPaint);

    // Top-Right
    canvas.drawLine(Offset(size.width - pad, pad), Offset(size.width - pad - corner, pad), bracketPaint);
    canvas.drawLine(Offset(size.width - pad, pad), Offset(size.width - pad, pad + corner), bracketPaint);

    // Bottom-Left
    canvas.drawLine(Offset(pad, size.height - pad), Offset(pad + corner, size.height - pad), bracketPaint);
    canvas.drawLine(Offset(pad, size.height - pad), Offset(pad, size.height - pad - corner), bracketPaint);

    // Bottom-Right
    canvas.drawLine(Offset(size.width - pad, size.height - pad), Offset(size.width - pad - corner, size.height - pad), bracketPaint);
    canvas.drawLine(Offset(size.width - pad, size.height - pad), Offset(size.width - pad, size.height - pad - corner), bracketPaint);

    // Scanning Line (only when not showing live video to avoid visual distraction)
    if (!hasLiveVideo) {
      final scanY = pad + (size.height - (pad * 2)) * scanProgress;
      final scanPaint = Paint()
        ..shader = LinearGradient(
          colors: [
            Colors.transparent,
            const Color(0xFF00D4FF).withValues(alpha: 0.4),
            const Color(0xFF00D4FF).withValues(alpha: 0.8),
            const Color(0xFF00D4FF).withValues(alpha: 0.4),
            Colors.transparent,
          ],
        ).createShader(Rect.fromLTWH(0, scanY, size.width, 2.0))
        ..strokeWidth = 1.5;

      canvas.drawLine(Offset(pad, scanY), Offset(size.width - pad, scanY), scanPaint);
    }
  }

  @override
  bool shouldRepaint(covariant _CockpitReticlePainter oldDelegate) =>
      oldDelegate.scanProgress != scanProgress ||
      oldDelegate.isPrivacy != isPrivacy ||
      oldDelegate.hasLiveVideo != hasLiveVideo;
}

  String _triggerDeviceState = 'on';
  final List<int> _selectedWeekdays = [1, 2, 3, 4, 5, 6, 7];

  @override
  void initState() {
    super.initState();
    final rule = widget.existingRule;
    if (rule != null) {
      _nameCtrl.text = rule.name;
      _selectedTriggerKind = rule.trigger.kind;
      if (rule.trigger.kind == TriggerKind.timeSchedule) {
        final parts = rule.trigger.value.split(':');
        if (parts.length == 2) {
          _selectedTime = TimeOfDay(
            hour: int.tryParse(parts[0]) ?? 22,
            minute: int.tryParse(parts[1]) ?? 0,
          );
        }
      } else if (rule.trigger.kind == TriggerKind.solarEvent) {
        _solarEvent = rule.trigger.value;
      } else if (rule.trigger.kind == TriggerKind.wifiPresence) {
        _presenceEvent = rule.trigger.value;
      } else if (rule.trigger.kind == TriggerKind.deviceState) {
        _triggerDeviceId = rule.trigger.sourceDeviceId;
        _triggerDeviceState = rule.trigger.value;
      }
      _selectedWeekdays
        ..clear()
        ..addAll(rule.trigger.weekdays);

      for (final p in rule.preconditions) {
        _selectedPreconditions.add(p.kind);
      }

      for (final a in rule.actions) {
        _actionSteps.add(
          _EditableActionStep(
            targetDeviceId: a.targetDeviceId,
            targetDeviceName: a.targetDeviceName,
            turnOn: a.turnOn,
            targetLevel: a.targetLevel,
            delaySecondsBefore: a.delaySecondsBefore,
          ),
        );
      }
    } else if (widget.initialTemplate != null) {
      WidgetsBinding.instance.addPostFrameCallback((_) {
        if (!mounted) return;
        final home = context.read<RealHomeProvider>();
        if (widget.initialTemplate == 'washer_finished' ||

  void _saveRule(BuildContext context) async {
    final name = _nameCtrl.text.trim().isEmpty ? 'Smart Routine' : _nameCtrl.text.trim();
    final home = context.read<RealHomeProvider>();
    final devices = home.devices;

    // Assemble Trigger
    String trigVal = '';
    String? srcDevId;
    String? srcDevName;

    switch (_selectedTriggerKind) {
      case TriggerKind.timeSchedule:
        trigVal = '${_selectedTime.hour.toString().padLeft(2, '0')}:${_selectedTime.minute.toString().padLeft(2, '0')}';
        break;
      case TriggerKind.solarEvent:
        trigVal = _solarEvent;
        break;
      case TriggerKind.wifiPresence:
        trigVal = _presenceEvent;
        break;
      case TriggerKind.deviceState:
        srcDevId = _triggerDeviceId ?? (devices.isNotEmpty ? devices.first.id : '');
        final matched = devices.where((d) => d.id == srcDevId);
        srcDevName = matched.isNotEmpty ? matched.first.name : 'Device';
        trigVal = _triggerDeviceState;
        break;
      default:
        trigVal = '22:00';
    }

    final trigger = SmartTrigger(
      kind: _selectedTriggerKind,
      value: trigVal,
      sourceDeviceId: srcDevId,
      sourceDeviceName: srcDevName,
      weekdays: _selectedWeekdays,
    );

    // Assemble Preconditions
    final preconditions = _selectedPreconditions.map((k) => SmartPrecondition(kind: k)).toList();

    // Assemble Actions
    final actions = _actionSteps.map((e) {
      return SmartActionStep(
        targetDeviceId: e.targetDeviceId,
        targetDeviceName: e.targetDeviceName,
        turnOn: e.turnOn,
        targetLevel: e.targetLevel,
        delaySecondsBefore: e.delaySecondsBefore,
      );
    }).toList();

    final ruleId = widget.existingRule?.id ?? 'rule_${DateTime.now().millisecondsSinceEpoch}';

    final rule = MultiConditionAutomationRule(
      id: ruleId,
      name: name,
      isEnabled: true,
      trigger: trigger,
      preconditions: preconditions,
      actions: actions,
    );

    if (widget.existingRule != null) {
      await home.updateMultiConditionRule(rule);
    } else {
      await home.addMultiConditionRule(rule);
    }

    HapticFeedback.heavyImpact();
    if (context.mounted) {
      Navigator.of(context).pop();
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          backgroundColor: const Color(0xFF30D158),
          content: Text('Saved and deployed "$name" to real hardware!'),
        ),
      );
    }
  }
}

  ParsedQrDevice? _decodeQrPayload(String rawText, [String? forcedProvider]) {
    final text = rawText.trim();
    if (text.isEmpty) return null;

    final providerKey = (forcedProvider != null && forcedProvider != 'all')
        ? forcedProvider
        : _detectProviderFromPayload(text);

    switch (providerKey) {
      case 'samsung_smartthings':
        return _parseSamsungSmartThingsQr(text);

      case 'lg_thinq':
        return _parseLgThinqQr(text);

      case 'wipro_smart_next':
      case 'wipro_tuya':
        return _parseWiproTuyaQr(text);

      case 'philips_hue':
        return _parsePhilipsHueQr(text);

      case 'philips_wiz':
        return _parsePhilipsWizQr(text);

      case 'realme_camera':
        return _parseRealmeCameraQr(text);

      case 'philips_camera':
        return _parsePhilipsCameraQr(text);

      case 'tplink_tapo':
        return _parseTapoQr(text);

      case 'home_assistant':
      default:
        // Try Matter Standard QR if it starts with MT:
        if (text.startsWith('MT:') || text.startsWith('CHIP:')) {
          return _parseMatterQr(text);
        }
        return _parseGenericOrMatterQr(text);
    }
  }

  String _detectProviderFromPayload(String text) {
    final lower = text.toLowerCase();
    if (lower.contains('smartthings') || lower.contains('samsung') || lower.contains('ecobubble') || lower.contains('windfree')) {
      return 'samsung_smartthings';
    }
    if (lower.contains('thinq') || lower.contains('lg_') || lower.contains('washtower') || lower.contains('dualcool') || lower.contains('instaview') || lower.contains('aerotower')) {
      return 'lg_thinq';
    }
    if (lower.contains('wipro') || lower.contains('smartlife') || lower.contains('tuya')) {
      return 'wipro_smart_next';
    }
    if (lower.contains('meethue') || lower.contains('philips-hue') || lower.contains('hue_bridge')) {
      return 'philips_hue';
    }
    if (lower.contains('wiz') || lower.contains('wizconnected')) {
      return 'philips_wiz';
    }
    if (lower.contains('realme') || lower.contains('rtsp://') && lower.contains('554')) {
      return 'realme_camera';
    }
    if (lower.contains('philips') && (lower.contains('cam') || lower.contains('safety'))) {
      return 'philips_camera';
    }
    if (lower.contains('tapo') || lower.contains('kasa') || lower.contains('tp-link')) {
      return 'tplink_tapo';
    }
    if (lower.startsWith('mt:') || lower.contains('matter') || lower.contains('homeassistant')) {
      return 'home_assistant';
    }
    return 'all';
  }

  // ---------------------------------------------------------------------------
  // SCAN OR TEST CODE HANDLERS
  // ---------------------------------------------------------------------------
  void _onQrDetected(String rawCode) {
    HapticFeedback.mediumImpact();
    final parsed = _decodeQrPayload(rawCode, _selectedProvider);
    if (parsed != null) {
      setState(() {
        _isScanning = false;
        _detectedDevice = parsed;
        _scanStatusText = 'Target Recognized: ${parsed.device.name}';
      });
    } else {
      setState(() {
        _scanStatusText = 'Unrecognized QR code format. Try again or paste code.';
      });
    }
  }

  Future<void> _pairDetectedDevice() async {
    if (_detectedDevice == null) return;
    setState(() => _isPairing = true);
    HapticFeedback.heavyImpact();

    final home = context.read<RealHomeProvider>();
    home.addDeviceDirect(_detectedDevice!.device);

    await Future.delayed(const Duration(milliseconds: 600));
    if (mounted) {
      setState(() {
        _isPairing = false;
        _pairingSuccessMsg = 'Successfully Paired ${_detectedDevice!.device.name}!';
      });
      await Future.delayed(const Duration(milliseconds: 900));
      if (mounted) {
        Navigator.pop(context);
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            backgroundColor: const Color(0xFF1C1C1E),
            content: Row(
              children: [
                Icon(Icons.check_circle_rounded, color: _detectedDevice!.brandColor),
                const SizedBox(width: 12),
                Expanded(
                  child: Text(
                    'Added ${_detectedDevice!.device.name} to ${_detectedDevice!.device.room} via ${_detectedDevice!.pairingProtocol}',
                    style: const TextStyle(fontWeight: FontWeight.w600, color: Colors.white),
                  ),
                ),
              ],
            ),
          ),
        );
      }
    }
  }

/// Custom Painter for 4 glowing reticle corner brackets
class _ReticleCornerPainter extends CustomPainter {
  final Color color;

  const _ReticleCornerPainter({required this.color});

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = color
      ..strokeWidth = 3.5
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round;

    const cornerLen = 28.0;

    // Top-Left
    canvas.drawLine(const Offset(0, 0), const Offset(cornerLen, 0), paint);
    canvas.drawLine(const Offset(0, 0), const Offset(0, cornerLen), paint);

    // Top-Right
    canvas.drawLine(Offset(size.width, 0), Offset(size.width - cornerLen, 0), paint);
    canvas.drawLine(Offset(size.width, 0), Offset(size.width, cornerLen), paint);

    // Bottom-Left
    canvas.drawLine(Offset(0, size.height), Offset(cornerLen, size.height), paint);
    canvas.drawLine(Offset(0, size.height), Offset(0, size.height - cornerLen), paint);

    // Bottom-Right
    canvas.drawLine(Offset(size.width, size.height), Offset(size.width - cornerLen, size.height), paint);
    canvas.drawLine(Offset(size.width, size.height), Offset(size.width, size.height - cornerLen), paint);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class _TeslaPowerTopologyPainter extends CustomPainter {
  final double totalWatts;
  final double hvacWatts;
  final double lightingMediaWatts;
  final int onlineCount;
  final int totalCount;
  final double progress;

  _TeslaPowerTopologyPainter({
    required this.totalWatts,
    required this.hvacWatts,
    required this.lightingMediaWatts,
    required this.onlineCount,
    required this.totalCount,
    required this.progress,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final topNode = Offset(size.width * 0.50, 22);
    final centerNode = Offset(size.width * 0.50, size.height * 0.52);
    final leftNode = Offset(size.width * 0.22, size.height - 24);
    final rightNode = Offset(size.width * 0.78, size.height - 24);

    final isFlowing = totalWatts > 0;

    // 1. Spline 1: Top (Main Grid) -> Center (Smart Home Bus)
    final pathTop = Path()
      ..moveTo(topNode.dx, topNode.dy)
      ..cubicTo(
        topNode.dx,
        topNode.dy + (centerNode.dy - topNode.dy) * 0.5,
        centerNode.dx,
        topNode.dy + (centerNode.dy - topNode.dy) * 0.5,
        centerNode.dx,
        centerNode.dy,
      );

    canvas.drawPath(
      pathTop,
      Paint()
        ..color = isFlowing
            ? const Color(0xFF30D158).withValues(alpha: 0.45)
            : Colors.white12
        ..style = PaintingStyle.stroke
        ..strokeWidth = isFlowing ? 2.5 : 1.5,
    );

    // 2. Spline 2: Center -> Left (Heavy Appliances: Washer & AC)
    final pathLeft = Path()
      ..moveTo(centerNode.dx, centerNode.dy)
      ..cubicTo(
        centerNode.dx - 30,
        centerNode.dy + 15,
        leftNode.dx + 25,
        leftNode.dy - 15,
        leftNode.dx,
        leftNode.dy,
      );

    canvas.drawPath(
      pathLeft,
      Paint()
        ..color = hvacWatts > 0
            ? const Color(0xFF00D4FF).withValues(alpha: 0.45)
            : Colors.white12
        ..style = PaintingStyle.stroke
        ..strokeWidth = hvacWatts > 0 ? 2.2 : 1.5,
    );

    // 3. Spline 3: Center -> Right (Lighting & Media)
    final pathRight = Path()
      ..moveTo(centerNode.dx, centerNode.dy)
      ..cubicTo(
        centerNode.dx + 30,
        centerNode.dy + 15,
        rightNode.dx - 25,
        rightNode.dy - 15,
        rightNode.dx,
        rightNode.dy,
      );

    canvas.drawPath(
      pathRight,
      Paint()
        ..color = lightingMediaWatts > 0
            ? const Color(0xFFBF5AF2).withValues(alpha: 0.45)
            : Colors.white12
        ..style = PaintingStyle.stroke
        ..strokeWidth = lightingMediaWatts > 0 ? 2.2 : 1.5,
    );

    // 4. Flowing Energy Particles (Velocity scaled by totalLiveWatts)
    if (isFlowing) {
      _drawParticlesOnPath(canvas, pathTop, progress, 4, const Color(0xFF30D158));

      if (hvacWatts > 0) {
        _drawParticlesOnPath(canvas, pathLeft, progress, 3, const Color(0xFF00D4FF));
      }
      if (lightingMediaWatts > 0) {
        _drawParticlesOnPath(canvas, pathRight, progress, 3, const Color(0xFFBF5AF2));
      }
    }

    // 5. Render Node Halos and Core Rings
    _drawNode(
      canvas,
      topNode,
      const Color(0xFF30D158),
      'MAIN GRID',
      '${totalWatts.round()} W',
      pulsing: isFlowing,
    );

    _drawNode(
      canvas,
      centerNode,
      const Color(0xFF0A84FF),
      'SMART BUS',
      '$onlineCount/$totalCount ONLINE',
      radius: 14,
    );

    _drawNode(
      canvas,
      leftNode,
      hvacWatts > 0 ? const Color(0xFF00D4FF) : Colors.white30,
      'HVAC / LAUNDRY',
      '${hvacWatts.round()} W',
      radius: 13,
    );

    _drawNode(
      canvas,
      rightNode,
      lightingMediaWatts > 0 ? const Color(0xFFBF5AF2) : Colors.white30,
      'LIGHTS / MEDIA',
      '${lightingMediaWatts.round()} W',
      radius: 13,
    );
  }

  void _drawParticlesOnPath(
    Canvas canvas,
    Path path,
    double progress,
    int count,
    Color particleColor,
  ) {
    final metrics = path.computeMetrics().toList();
    if (metrics.isEmpty) return;
    final metric = metrics.first;

    for (int p = 0; p < count; p++) {
      final t = (progress + (p / count)) % 1.0;
      final tangent = metric.getTangentForOffset(metric.length * t);
      if (tangent != null) {
        // Outer soft glow
        canvas.drawCircle(
          tangent.position,
          5.5,
          Paint()..color = particleColor.withValues(alpha: 0.25),
        );
        // Inner intense core
        canvas.drawCircle(
          tangent.position,
          2.6,
          Paint()..color = Colors.white,
        );
      }
    }
  }

  void _drawNode(
    Canvas canvas,
    Offset center,
    Color tint,
    String label,
    String sublabel, {
    double radius = 15,
    bool pulsing = false,
  }) {
    final pulseScale = pulsing ? (1.0 + 0.15 * math.sin(progress * 2 * math.pi)) : 1.0;

    // Soft Ambient Glow
    canvas.drawCircle(
      center,
      radius * 1.8 * pulseScale,
      Paint()..color = tint.withValues(alpha: 0.18),
    );

    // Inner Shell
    canvas.drawCircle(
      center,
      radius,
      Paint()
        ..color = const Color(0xFF0B1220)
        ..style = PaintingStyle.fill,
    );
    canvas.drawCircle(
      center,
      radius,
      Paint()
        ..color = tint
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2.0,
    );

    // Core Dot
    canvas.drawCircle(
      center,
      4.0,
      Paint()..color = tint,
    );

    // Text Badges
    final textPainter = TextPainter(
      text: TextSpan(
        children: [
          TextSpan(
            text: '$label\n',
            style: TextStyle(
              fontSize: 7.5,
              fontWeight: FontWeight.w700,
              letterSpacing: 0.6,
              color: tint.withValues(alpha: 0.90),
            ),
          ),
          TextSpan(
            text: sublabel,
            style: const TextStyle(
              fontSize: 8.5,
              fontWeight: FontWeight.w800,
              color: Colors.white,
            ),
          ),
        ],
      ),
      textAlign: TextAlign.center,
      textDirection: TextDirection.ltr,
    )..layout();

    final isTop = center.dy < 40;
    final textOffset = Offset(
      center.dx - (textPainter.width / 2),
      isTop ? center.dy + radius + 4 : center.dy - radius - textPainter.height - 3,
    );
    textPainter.paint(canvas, textOffset);
  }

  @override
  bool shouldRepaint(covariant _TeslaPowerTopologyPainter oldDelegate) =>
      oldDelegate.progress != progress ||
      oldDelegate.totalWatts != totalWatts ||
      oldDelegate.hvacWatts != hvacWatts ||
      oldDelegate.lightingMediaWatts != lightingMediaWatts;
}
