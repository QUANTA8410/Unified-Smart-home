/// Credentials for the Tuya Cloud OpenAPI ("Smart Home" project linked to Smart Life).
class TuyaCredentials {
  final String accessId;
  final String accessSecret;
  final String region; // 'in', 'eu', 'eu-w', 'us', 'us-e', 'cn', 'sg'

  const TuyaCredentials({
    required this.accessId,
    required this.accessSecret,
    this.region = 'in', // Default to Indian region for Wipro Smart Next
  });

  bool get isConfigured =>
      accessId.trim().isNotEmpty &&
      accessSecret.trim().isNotEmpty &&
      TuyaService.hosts.containsKey(region);

  Map<String, dynamic> toJson() => {
        'accessId': accessId,
        'accessSecret': accessSecret,
        'region': region,
      };

  factory TuyaCredentials.fromJson(Map<String, dynamic> json) {
    return TuyaCredentials(
      accessId: json['accessId'] as String? ?? '',
      accessSecret: json['accessSecret'] as String? ?? '',
      region: json['region'] as String? ?? 'in',
    );
  }
}

/// Minimal client for the Tuya Cloud OpenAPI matching House Control's TuyaService.swift.
/// Covers device listing, live status reading, and command execution with HMAC-SHA256 signing.
class TuyaService {
  static const Map<String, String> hosts = {
    'in': 'openapi.tuyain.com', // India (Wipro Next primary region)
    'eu': 'openapi.tuyaeu.com', // Central Europe
    'eu-w': 'openapi-weaz.tuyaeu.com', // Western Europe
    'us': 'openapi.tuyaus.com', // Western America
    'us-e': 'openapi-ueaz.tuyaus.com', // Eastern America
    'cn': 'openapi.tuyacn.com', // China
    'sg': 'openapi-sg.iotbing.com', // Singapore
  };

  TuyaCredentials? credentials;
  String? _token;
  DateTime _tokenExpiry = DateTime.fromMillisecondsSinceEpoch(0);

  final Map<String, Set<String>> knownCodes = {};
  final Map<String, Map<String, List<double>>> knownRanges = {};

  TuyaService({this.credentials});

  bool get isConfigured => credentials?.isConfigured ?? false;

  Map<String, dynamic> get publicInfo => {
        'configured': isConfigured,
        'accessId': credentials?.accessId ?? '',
        'region': credentials?.region ?? 'in',
        'hasSecret': credentials?.accessSecret.isNotEmpty ?? false,
      };

  // MARK: - OpenAPI Calls

  /// Every device of the linked account, with the DP codes it supports.
  Future<List<Map<String, dynamic>>> listDevices() async {
    final List<Map<String, dynamic>> devices = [];
    String? lastKey;

    do {
      final query = <String, String>{'size': '50'};
      if (lastKey != null) query['last_row_key'] = lastKey;

      final result = await request('GET', '/v1.0/iot-01/associated-users/devices', query: query);
      if (result is! Map<String, dynamic>) break;

      final devList = result['devices'] as List? ?? [];
      for (final d in devList) {
        if (d is! Map<String, dynamic>) continue;
        final id = d['id'] as String?;
        if (id == null) continue;

        final status = d['status'] as List? ?? [];
        final codes = <String>{};
        for (final item in status) {
          if (item is Map && item['code'] != null) {
            codes.add(item['code'].toString());
          }
        }
        knownCodes[id] = codes;

        devices.add({
          'id': id,
          'name': d['name'] ?? id,
          'category': d['category'] ?? '',
          'product': d['product_name'] ?? '',
          'online': d['online'] ?? false,
          'codes': codes.toList(),
        });
      }

      final hasMore = result['has_more'] as bool? ?? false;
      lastKey = hasMore ? result['last_row_key'] as String? : null;
    } while (lastKey != null);

    return devices;
  }

  /// Current data points for the given devices, keyed by device id.
  Future<Map<String, List<Map<String, dynamic>>>> status(List<String> ids) async {
    final Map<String, List<Map<String, dynamic>>> out = {};

    for (var i = 0; i < ids.countChunks(20); i++) {
      final chunk = ids.skip(i * 20).take(20).toList();
      final result = await request(
        'GET',
        '/v1.0/iot-03/devices/status',
        query: {'device_ids': chunk.join(',')},
      );

      if (result is List) {
        for (final entry in result) {
          if (entry is! Map<String, dynamic>) continue;
          final id = entry['id'] as String?;
          final statusList = (entry['status'] as List?)?.whereType<Map<String, dynamic>>().toList() ?? [];
          if (id != null) {
            out[id] = statusList;
            knownCodes[id] = statusList.map((e) => e['code'].toString()).toSet();
          }
        }
      }
    }

    return out;
  }

  /// Sends standard instruction commands to a device via Tuya OpenAPI.
  Future<void> send(List<Map<String, dynamic>> commands, {required String toDeviceId}) async {
    if (commands.isEmpty) return;
    final body = jsonEncode({'commands': commands});
    await request('POST', '/v1.0/iot-03/devices/$toDeviceId/commands', body: utf8.encode(body));
  }

  // MARK: - HMAC-SHA256 Token & Request Signing

  Future<String> accessToken() async {
    if (_token != null && _tokenExpiry.isAfter(DateTime.now().add(const Duration(seconds: 60)))) {
      return _token!;
    }

    final result = await request('GET', '/v1.0/token', query: {'grant_type': '1'}, authorized: false);
    if (result is! Map<String, dynamic> || result['access_token'] == null) {
      throw Exception('Tuya did not return an access token: $result');
    }

    _token = result['access_token'].toString();
    final expireSec = (result['expire_time'] as num?)?.toInt() ?? 3600;
    _tokenExpiry = DateTime.now().add(Duration(seconds: expireSec));
    return _token!;
  }

  Future<dynamic> request(
    String method,
    String path, {
    Map<String, String> query = const {},
    List<int>? body,
    bool authorized = true,
    bool retry = true,
  }) async {
    final creds = credentials;
    if (creds == null || !creds.isConfigured) {
      throw Exception('Tuya credentials are not configured or region is invalid');
    }
    final host = hosts[creds.region];
    if (host == null) throw Exception('Unknown region: ${creds.region}');

    // Sort query alphabetically
    final sortedKeys = query.keys.toList()..sort();
    final queryString = sortedKeys.map((k) => '$k=${query[k]}').join('&');
    final signedUrl = queryString.isEmpty ? path : '$path?$queryString';
    final fullUri = Uri.parse('https://$host$signedUrl');

    final bodyBytes = body ?? [];
    final bodyHash = sha256.convert(bodyBytes).toString().toLowerCase();
    final t = '${DateTime.now().millisecondsSinceEpoch}';
    final token = authorized ? await accessToken() : '';

    // Authentic Tuya HMAC-SHA256 signature payload
    final payload = '${creds.accessId}$token$t$method\n$bodyHash\n\n$signedUrl';
    final mac = Hmac(sha256, utf8.encode(creds.accessSecret));
    final sign = mac.convert(utf8.encode(payload)).toString().toUpperCase();

    final headers = <String, String>{
      'client_id': creds.accessId,
      'sign': sign,
      't': t,
      'sign_method': 'HMAC-SHA256',
    };
    if (authorized) headers['access_token'] = token;
    if (body != null) headers['Content-Type'] = 'application/json';

    final http.Response res;
    if (method == 'GET') {
      res = await http.get(fullUri, headers: headers).timeout(const Duration(seconds: 12));
    } else if (method == 'POST') {
      res = await http.post(fullUri, headers: headers, body: body).timeout(const Duration(seconds: 12));
    } else if (method == 'PUT') {
      res = await http.put(fullUri, headers: headers, body: body).timeout(const Duration(seconds: 12));
    } else if (method == 'DELETE') {
      res = await http.delete(fullUri, headers: headers, body: body).timeout(const Duration(seconds: 12));
    } else {
      throw Exception('Unsupported HTTP method: $method');
    }

    final dynamic parsed;
    try {
      parsed = jsonDecode(res.body);
    } catch (_) {
      throw Exception('Unexpected response format from Tuya (${res.statusCode})');
    }

    if (parsed is Map<String, dynamic>) {
      if (parsed['success'] == true) return parsed['result'];
      final code = (parsed['code'] as num?)?.toInt() ?? 0;

      // 1010 / 1011: token expired or invalid. Re-authenticate once
      if (authorized && retry && (code == 1010 || code == 1011)) {
        _token = null;
        return request(method, path, query: query, body: body, authorized: true, retry: false);
      }
      throw Exception('${parsed['msg'] ?? 'Tuya request failed'} (code $code)');
    }
    return parsed;
  }
}

/// Translates between the app's device state and Tuya's standard instruction set.
/// Matches House Control's TuyaMapper.swift with full support for lights, ACs, and IR remotes.
class TuyaMapper {
  static const Map<String, String> acModes = {
    'cool': 'cold',
    'heat': 'hot',
    'dry': 'wet',
    'fan': 'wind',
    'auto': 'auto',
  };

  static const Map<String, String> acFans = {
    'auto': 'auto',
    'low': 'low',
    'medium': 'mid',
    'high': 'high',
    'turbo': 'strong',
  };

  static const Map<String, int> irModes = {
    'cool': 0,
    'heat': 1,
    'auto': 2,
    'fan': 3,
    'dry': 4,
  };

  static const Map<String, int> irFans = {
    'auto': 0,
    'low': 1,
    'medium': 2,
    'high': 3,
    'turbo': 3,
  };

  static String? pick(List<String> candidates, Set<String> codes, [String? override]) {
    if (override != null && override.isNotEmpty) return override;
    for (final c in candidates) {
      if (codes.contains(c)) return c;
    }
    return null;
  }

  static (double, double) range(String code, Map<String, List<double>> ranges, {double floor = 0}) {
    final r = ranges[code];
    if (r != null && r.length == 2 && r[1] > r[0]) {
      return (r[0], r[1]);
    }
    return code.endsWith('_v2') ? (floor > 0 ? 10.0 : 0.0, 1000.0) : (floor, 255.0);
  }

  /// Converts RGB hex e.g. '#FF5733' to HSV (h: 0..360, s: 0..1, v: 0..1)
  static (double, double, double)? hexToHsv(String hex) {
    var clean = hex.replaceAll('#', '').trim();
    if (clean.length == 6) {
      final r = int.parse(clean.substring(0, 2), radix: 16) / 255.0;
      final g = int.parse(clean.substring(2, 4), radix: 16) / 255.0;
      final b = int.parse(clean.substring(4, 6), radix: 16) / 255.0;

      final maxC = [r, g, b].reduce((curr, next) => curr > next ? curr : next);
      final minC = [r, g, b].reduce((curr, next) => curr < next ? curr : next);
      final delta = maxC - minC;

      double h = 0;
      if (delta > 0) {
        if (maxC == r) {
          h = ((g - b) / delta) % 6;
        } else if (maxC == g) {
          h = ((b - r) / delta) + 2;
        } else {
          h = ((r - g) / delta) + 4;
        }
        h *= 60;
        if (h < 0) h += 360;
      }
      final s = maxC == 0 ? 0.0 : delta / maxC;
      final v = maxC;
      return (h, s, v);
    }
    return null;
  }

  /// Commands to set light state (on/off, brightness, color temp Kelvin, RGB)
  static List<Map<String, dynamic>> lightCommands({
    required Map<String, dynamic> patch,
    Map<String, dynamic> state = const {},
    Map<String, dynamic> link = const {},
    required Set<String> codes,
    Map<String, List<double>> ranges = const {},
  }) {
    final custom = link['codes'] as Map<String, dynamic>? ?? {};
    final List<Map<String, dynamic>> out = [];

    if (patch.containsKey('on')) {
      final code = pick(['switch_led', 'switch_1', 'switch'], codes, custom['switch'] as String?) ?? 'switch_led';
      out.add({'code': code, 'value': patch['on'] == true});
    }

    if (patch.containsKey('brightness')) {
      final b = (patch['brightness'] as num).toDouble();
      final code = pick(['bright_value_v2', 'bright_value'], codes);
      if (code != null) {
        final (lo, hi) = range(code, ranges, floor: 25);
        out.add({'code': code, 'value': (lo + b * (hi - lo)).round()});
      }
    }

    if (patch.containsKey('temp')) {
      final k = (patch['temp'] as num).toDouble();
      final code = pick(['temp_value_v2', 'temp_value'], codes);
      if (code != null) {
        final t = ((k - 2200) / (6500 - 2200)).clamp(0.0, 1.0);
        final (lo, hi) = range(code, ranges);
        if (codes.contains('work_mode')) {
          out.add({'code': 'work_mode', 'value': 'white'});
        }
        out.add({'code': code, 'value': (lo + t * (hi - lo)).round()});
      }
    }

    if (patch.containsKey('color')) {
      final hex = patch['color'] as String;
      final hsv = hexToHsv(hex);
      final code = pick(['colour_data_v2', 'colour_data'], codes);
      if (hsv != null && code != null) {
        final scale = range(code == 'colour_data_v2' ? 'bright_value_v2' : 'bright_value', ranges).$2 > 255 ? 1000.0 : 255.0;
        final v = ((state['brightness'] as num?)?.toDouble() ?? 1.0).clamp(0.01, 1.0);
        if (codes.contains('work_mode')) {
          out.add({'code': 'work_mode', 'value': 'colour'});
        }
        out.add({
          'code': code,
          'value': jsonEncode({
            'h': hsv.$1.round(),
            's': (hsv.$2 * scale).round(),
            'v': (v * scale).round(),
          }),
        });
      }
    }

    return out;
  }

  /// Parses device status list into uniform light state
  static Map<String, dynamic> lightState(List<Map<String, dynamic>> status, {Map<String, List<double>> ranges = const {}}) {
    final Map<String, dynamic> out = {};
    for (final dp in status) {
      final code = dp['code'] as String?;
      final value = dp['value'];
      if (code == null || value == null) continue;

      switch (code) {
        case 'switch_led':
        case 'switch_1':
        case 'switch':
          if (!out.containsKey('on') && value is bool) out['on'] = value;
          break;
        case 'bright_value_v2':
        case 'bright_value':
          if (value is num) {
            final (lo, hi) = range(code, ranges, floor: 25);
            out['brightness'] = ((value - lo) / (hi - lo)).clamp(0.05, 1.0);
          }
          break;
        case 'temp_value_v2':
        case 'temp_value':
          if (value is num) {
            final (lo, hi) = range(code, ranges);
            out['temp'] = 2200 + ((value - lo) / (hi - lo)).clamp(0.0, 1.0) * 4300;
          }
          break;
      }
    }
    return out;
  }

  /// True for an AC that is controlled via infrared blaster
  static bool isInfraredAC(Set<String> codes) => codes.contains('PowerOn') || codes.contains('PowerOff');

  /// Key presses for infrared remote
  static List<Map<String, dynamic>> infraredKeys(Map<String, dynamic> patch) {
    final List<Map<String, dynamic>> out = [];
    if (patch.containsKey('on')) {
      final key = patch['on'] == true ? 'PowerOn' : 'PowerOff';
      out.add({'code': key, 'value': key});
    }
    if (patch.containsKey('mode') && irModes.containsKey(patch['mode'])) {
      out.add({'code': 'M', 'value': irModes[patch['mode']]});
    }
    if (patch.containsKey('target')) {
      out.add({'code': 'T', 'value': (patch['target'] as num).round()});
    }
    if (patch.containsKey('fan') && irFans.containsKey(patch['fan'])) {
      out.add({'code': 'F', 'value': irFans[patch['fan']]});
    }
    return out;
  }

  /// AC unit commands
  static List<Map<String, dynamic>> acCommands({
    required Map<String, dynamic> patch,
    Map<String, dynamic> link = const {},
    required Set<String> codes,
  }) {
    if (isInfraredAC(codes)) return infraredKeys(patch);

    final custom = link['codes'] as Map<String, dynamic>? ?? {};
    final List<Map<String, dynamic>> out = [];

    if (patch.containsKey('on')) {
      final code = pick(['switch', 'power', 'switch_1'], codes, custom['switch'] as String?) ?? 'switch';
      out.add({'code': code, 'value': patch['on'] == true});
    }
    if (patch.containsKey('target')) {
      final code = pick(['temp_set'], codes, custom['target'] as String?) ?? 'temp_set';
      out.add({'code': code, 'value': (patch['target'] as num).round()});
    }
    if (patch.containsKey('mode') && acModes.containsKey(patch['mode'])) {
      final code = pick(['mode'], codes, custom['mode'] as String?) ?? 'mode';
      out.add({'code': code, 'value': acModes[patch['mode']]});
    }
    if (patch.containsKey('fan') && acFans.containsKey(patch['fan'])) {
      final code = pick(['fan_speed_enum', 'windspeed'], codes, custom['fan'] as String?);
      if (code != null) out.add({'code': code, 'value': acFans[patch['fan']]});
    }
    if (patch.containsKey('sleep') && codes.contains('sleep')) {
      out.add({'code': 'sleep', 'value': patch['sleep'] == true});
    }
    if (patch.containsKey('eco') && codes.contains('eco')) {
      out.add({'code': 'eco', 'value': patch['eco'] == true});
    }
    return out;
  }

  /// Parses status list into uniform AC state
  static Map<String, dynamic> acState(List<Map<String, dynamic>> status) {
    final Map<String, dynamic> out = {};
    for (final dp in status) {
      final code = dp['code'] as String?;
      final value = dp['value'];
      if (code == null || value == null) continue;

      switch (code) {
        case 'switch':
        case 'power':
          if (value is bool) out['on'] = value;
          break;
        case 'temp_set':
          if (value is num) out['target'] = value.toDouble();
          break;
        case 'temp_current':
          if (value is num) out['roomTemp'] = value.toDouble();
          break;
        case 'mode':
          if (value is String) {
            for (final entry in acModes.entries) {
              if (entry.value == value) out['mode'] = entry.key;
            }
          }
          break;
        case 'fan_speed_enum':
        case 'windspeed':
          if (value is String) {
            for (final entry in acFans.entries) {
              if (entry.value == value) out['fan'] = entry.key;
            }
          }
          break;
      }
    }
    return out;
  }
}

  int countChunks(int chunkSize) => (length / chunkSize).ceil();
}

/// Hybrid Wipro / Tuya Direct Local Encrypted Frame & Cloud Service.
class WiproTuyaService {
  final TuyaService cloud = TuyaService();

  /// Test whether Wipro bulb is reachable at the given LAN IP address
  Future<bool> checkConnectivity(String host, {Duration timeout = const Duration(seconds: 2)}) async {
    if (host.isEmpty) return false;
    try {
      final socket = await Socket.connect(host, 6668, timeout: timeout);
      socket.destroy();
      return true;
    } catch (_) {
      try {
        final socket = await Socket.connect(host, 80, timeout: timeout);
        socket.destroy();
        return true;
      } catch (_) {}
    }
    return false;
  }

  /// Sends raw Tuya 3.3 / 3.4 LAN DPS payload to the Wipro bulb over TCP port 6668
  Future<bool> sendLocalCommand({
    required String host,
    required Map<String, dynamic> dps,
    String? localKey,
    String? deviceId,
  }) async {
    if (host.isEmpty) return true;

    // If a 16-character localKey is available, send authentic AES-128 encrypted frame
    if (localKey != null && localKey.length == 16) {
      return sendEncryptedFrame(
        host: host,
        deviceId: deviceId ?? 'wipro_bulb',
        localKey: localKey,
        dps: dps,
      );
    }

    try {
      final socket = await Socket.connect(host, 6668, timeout: const Duration(seconds: 1));
      final payload = jsonEncode({
        'gwId': deviceId ?? 'wipro_bulb',
        'devId': deviceId ?? 'wipro_bulb',
        'uid': 'wipro_user',
        't': '${DateTime.now().millisecondsSinceEpoch ~/ 1000}',
        'dps': dps,
      });

      socket.write(payload);
      await socket.flush();
      socket.destroy();
      return true;
    } catch (_) {
      return false;
    }
  }

  /// Direct Tuya / Smart Life local encrypted frame transmission (v3.3 AES-128-ECB PKCS7 + CRC32)
  Future<bool> sendEncryptedFrame({
    required String host,
    required String deviceId,
    required String localKey,
    required Map<String, dynamic> dps,
  }) async {
    if (host.isEmpty || localKey.length != 16) return false;

    try {
      final socket = await Socket.connect(host, 6668, timeout: const Duration(seconds: 3));
      try {
        final payload = jsonEncode({
          'gwId': deviceId,
          'devId': deviceId,
          'uid': deviceId,
          't': (DateTime.now().millisecondsSinceEpoch ~/ 1000).toString(),
          'dps': dps,
        });

        final key = enc.Key.fromUtf8(localKey);
        final encrypter = enc.Encrypter(enc.AES(key, mode: enc.AESMode.ecb, padding: 'PKCS7'));
        final encrypted = encrypter.encrypt(payload).bytes;

        final body = BytesBuilder()
          ..add(utf8.encode('3.3\x00\x00\x00\x00\x00\x00\x00\x00\x00\x00\x00\x00'))
          ..add(encrypted);
        final bodyBytes = body.toBytes();

        final header = ByteData(16)
          ..setUint32(0, 0x000055AA, Endian.big)
          ..setUint32(4, 1, Endian.big)
          ..setUint32(8, 7, Endian.big) // 7 = CONTROL
          ..setUint32(12, bodyBytes.length + 8, Endian.big);

        final frame = BytesBuilder()
          ..add(header.buffer.asUint8List())
          ..add(bodyBytes);

        // IEEE 802.3 CRC32
        int crc = 0xFFFFFFFF;
        for (final b in frame.toBytes()) {
          crc ^= b;
          for (int i = 0; i < 8; i++) {
            crc = (crc & 1) != 0 ? (crc >> 1) ^ 0xEDB88320 : crc >> 1;
          }
        }
        crc ^= 0xFFFFFFFF;

        final footer = ByteData(8)
          ..setUint32(0, crc, Endian.big)
          ..setUint32(4, 0x0000AA55, Endian.big);
        frame.add(footer.buffer.asUint8List());

        socket.add(frame.toBytes());
        await socket.flush();
        return true;
      } finally {
        await socket.close();
      }
    } catch (_) {
      return false;
    }
  }

  /// Power toggle (DPS 20: boolean)
  Future<bool> setPower({
    required String host,
    required bool isPowered,
    String? cloudDeviceId,
    String? localKey,
  }) async {
    return sendLocalCommand(
      host: host,
      dps: {'20': isPowered},
      localKey: localKey,
      deviceId: cloudDeviceId,
    );
  }

  /// Brightness: 0.0 - 1.0 mapped to Tuya DPS 22 (10 - 1000)
  Future<bool> setBrightness({
    required String host,
    required double brightness,
    String? cloudDeviceId,
    String? localKey,
  }) async {
    final tuyaVal = (brightness * 1000).clamp(10, 1000).round();
    return sendLocalCommand(
      host: host,
      dps: {'22': tuyaVal},
      localKey: localKey,
      deviceId: cloudDeviceId,
    );
  }

  /// Color Temperature: 2700K - 6500K mapped to Tuya DPS 23 (0 - 1000)
  Future<bool> setColorTemp({
    required String host,
    required int kelvin,
    String? cloudDeviceId,
    String? localKey,
  }) async {
    final percent = ((kelvin - 2700) / (6500 - 2700)).clamp(0.0, 1.0);
    final tuyaVal = (percent * 1000).round();
    return sendLocalCommand(
      host: host,
      dps: {'23': tuyaVal},
      localKey: localKey,
      deviceId: cloudDeviceId,
    );
  }

  /// RGB Color: mapped to Tuya DPS 24 (HSV hex string)
  Future<bool> setColorHex({
    required String host,
    required String hexRgb,
    String? cloudDeviceId,
    String? localKey,
  }) async {
    return sendLocalCommand(
      host: host,
      dps: {'24': hexRgb, '21': 'colour'},
      localKey: localKey,
      deviceId: cloudDeviceId,
    );
  }

  /// Listens on UDP port 6667 for Tuya / Smart Life local LAN broadcast beacons
  Future<List<Map<String, dynamic>>> discoverLocalTuyaDevices({
    Duration timeout = const Duration(seconds: 2),
  }) async {
    final List<Map<String, dynamic>> discovered = [];
    RawDatagramSocket? socket;
    try {
      socket = await RawDatagramSocket.bind(
        InternetAddress.anyIPv4,
        6667,
        reuseAddress: true,
        reusePort: false,
      );
      socket.broadcastEnabled = true;

      final completer = Completer<List<Map<String, dynamic>>>();
      final sub = socket.listen((RawSocketEvent event) {
        if (event == RawSocketEvent.read) {
          final datagram = socket?.receive();
          if (datagram != null) {
            try {
              final text = utf8.decode(datagram.data, allowMalformed: true);
              if (text.contains('gwId') || text.contains('ip')) {
                discovered.add({
                  'ip': datagram.address.address,
                  'port': datagram.port,
                  'raw': text,
                  'discoveredAt': DateTime.now().toIso8601String(),
                });
              }
            } catch (_) {}
          }
        }
      });

      Timer(timeout, () {
        sub.cancel();
        socket?.close();
        if (!completer.isCompleted) completer.complete(discovered);
      });

      return await completer.future;
    } catch (_) {
      socket?.close();
      return discovered;
    }
  }
}

class _AppleSpecularBorderPainter extends CustomPainter {
  final double radius;
  final bool isActive;
  final Color tint;

  _AppleSpecularBorderPainter({
    required this.radius,
    required this.isActive,
    required this.tint,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final rect = Offset.zero & size;
    final rrect = RRect.fromRectAndRadius(rect.deflate(0.6), Radius.circular(radius));
    final paint = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.25
      ..shader = LinearGradient(
        begin: Alignment.topLeft,
        end: Alignment.bottomRight,
        colors: [
          Colors.white.withValues(alpha: isActive ? 0.75 : 0.42),
          isActive ? tint.withValues(alpha: 0.45) : Colors.white.withValues(alpha: 0.10),
          Colors.white.withValues(alpha: 0.03),
          Colors.white.withValues(alpha: 0.22),
        ],
        stops: const [0.0, 0.35, 0.7, 1.0],
      ).createShader(rect);

    canvas.drawRRect(rrect, paint);
  }

  @override
  bool shouldRepaint(covariant _AppleSpecularBorderPainter old) =>
      old.isActive != isActive || old.tint != tint;
}

  Future<void> _applyScene(BuildContext context, RealHomeProvider home, String sceneId) async {
    switch (sceneId) {
      case 'Bright':
        for (final d in home.devices) {
          if (d.category == 'light') {
            if (!d.isOn) await home.toggleDevicePower(d);
            await home.setDeviceLevel(d, 100.0);
          } else if (d.category == 'ac') {
            if (!d.isOn) await home.toggleDevicePower(d);
            await home.setDeviceLevel(d, 22.0);
          }
        }
        if (context.mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('☀️ Scene: Bright 100% Warm White & 22°C Cool')),
          );
        }
        break;
      case 'Cozy':
        for (final d in home.devices) {
          if (d.category == 'light') {
            if (!d.isOn) await home.toggleDevicePower(d);
            await home.setDeviceLevel(d, 60.0);
          } else if (d.category == 'ac') {
            if (!d.isOn) await home.toggleDevicePower(d);
            await home.setDeviceLevel(d, 24.0);
          }
        }
        if (context.mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('🕯️ Scene: Cozy 60% Warm Amber & 24°C')),
          );
        }
        break;
      case 'Movie':
        for (final d in home.devices) {
          if (d.category == 'tv') {
            if (!d.isOn) await home.toggleDevicePower(d);
          } else if (d.category == 'light') {
            if (d.name.toLowerCase().contains('backlight') ||
                d.name.toLowerCase().contains('ambilight') ||
                d.name.toLowerCase().contains('wipro')) {
              if (!d.isOn) await home.toggleDevicePower(d);
              await home.setDeviceLevel(d, 40.0);
            } else if (d.isOn) {
              await home.toggleDevicePower(d);
            }
          }
        }
        if (context.mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('🎬 Scene: Movie Cinema Mode')),
          );
        }
        break;
      case 'Night':
        for (final d in home.devices) {
          if (d.category == 'tv' && d.isOn) {
            await home.toggleDevicePower(d);
          } else if (d.category == 'light') {
            if (d.name.toLowerCase().contains('bed') ||
                d.name.toLowerCase().contains('night')) {
              if (!d.isOn) await home.toggleDevicePower(d);
              await home.setDeviceLevel(d, 15.0);
            } else if (d.isOn) {
              await home.toggleDevicePower(d);
            }
          } else if (d.category == 'ac') {
            if (!d.isOn) await home.toggleDevicePower(d);
            await home.setDeviceLevel(d, 25.0);
          }
        }
        if (context.mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('🌙 Scene: Night Light 15% & AC 25°C Sleep')),
          );
        }
        break;
      case 'Away':
        for (final d in home.devices) {
          if (d.category == 'light' && d.isOn) {
            await home.toggleDevicePower(d);
          } else if (d.category == 'tv' && d.isOn) {
            await home.toggleDevicePower(d);
          } else if (d.category == 'ac' && d.isOn) {
            await home.toggleDevicePower(d);
          }
        }
        if (context.mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('🛡️ Scene: Away Mode • Devices Off • Security Armed')),
          );
        }
        break;
    }
  }

      if (roomId != 'all' && fix.roomId.toLowerCase() != roomId.toLowerCase()) continue;
      if (fix.linkedRealDeviceId != null) {
        final dev = home.devices.where((d) => d.id == fix.linkedRealDeviceId).firstOrNull;
        if (dev?.isOn == true) count++;
      }
    }
    return count;
  }

  Future<void> _applyHouseControlScene(HouseScene3D scene, RealHomeProvider home) async {
    HapticFeedback.mediumImpact();
    ScaffoldMessenger.of(context).clearSnackBars();
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Row(
          children: [
            Icon(_iconForScene(scene.icon), color: const Color(0xFFFF9F0A), size: 18),
            const SizedBox(width: 10),
            Text('Scene: "${scene.name}" applied', style: const TextStyle(fontWeight: FontWeight.bold)),
          ],
        ),
        backgroundColor: const Color(0xFF1E232E),
        duration: const Duration(seconds: 2),
      ),
    );

    // Apply scene to linked lights
    final starConfig = scene.lights['*'] as Map<String, dynamic>?;
    final bool? starOn = starConfig?['on'] as bool?;
    final double? starBrightness = (starConfig?['brightness'] as num?)?.toDouble();

    for (final fix in _layout.fixtures) {
      if (fix.fixtureType == 'wall_ac') continue;
      final specific = scene.lights[fix.id] as Map<String, dynamic>?;
      final targetOn = specific?['on'] as bool? ?? starOn;
      final targetBrightness = (specific?['brightness'] as num?)?.toDouble() ?? starBrightness;

      if (fix.linkedRealDeviceId != null) {
        final dev = home.devices.where((d) => d.id == fix.linkedRealDeviceId).firstOrNull;
        if (dev != null) {
          if (targetOn != null && dev.isOn != targetOn) {
            await home.toggleDevicePower(dev);
          }
          if (targetBrightness != null && (targetOn ?? true)) {
            await home.setDeviceLevel(dev, (targetBrightness * 100).clamp(1.0, 100.0));
          }
        }
      }
    }

    // Apply scene to linked ACs
    final starAc = scene.acs['*'] as Map<String, dynamic>?;
    final bool? starAcOn = starAc?['on'] as bool?;

    for (final fix in _layout.fixtures) {
      if (fix.fixtureType != 'wall_ac') continue;
      final specificAc = scene.acs[fix.id] as Map<String, dynamic>?;
      final targetAcOn = specificAc?['on'] as bool? ?? starAcOn;

      if (fix.linkedRealDeviceId != null) {
        final dev = home.devices.where((d) => d.id == fix.linkedRealDeviceId).firstOrNull;
        if (dev != null && targetAcOn != null && dev.isOn != targetAcOn) {
          await home.toggleDevicePower(dev);
        }
      }
    }
  }

  Future<void> _bulkToggleAll(bool turnOn, RealHomeProvider home) async {
    HapticFeedback.heavyImpact();

          content: Text(turnOn ? 'All Devices Turned On' : 'All Devices Turned Off'),
          backgroundColor: turnOn ? const Color(0xFF30D158) : const Color(0xFFFF453A),
        ),
      );
    }

  Future<void> _pairNow(BuildContext context, RealHomeProvider home) async {
    setState(() => _statusMsg = 'Connecting to real device endpoint...');

      if (_mode == 1) {
        final count = await home.linkSmartThingsAccount(_tokenCtrl.text);

          Navigator.pop(context);
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('Imported $count real SmartThings devices!')),
          );
        }
      } else if (_mode == 2) {
        final count = await home.linkLgThinqAccount(
          _tokenCtrl.text,
          countryCode: _countryCtrl.text.trim().isEmpty ? 'IN' : _countryCtrl.text.trim(),
        );
        if (context.mounted) {
          Navigator.pop(context);
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('Imported $count real LG ThinQ devices!')),
          );
        }
      } else if (_mode == 3) {
        final count = await home.autoHarvestWiproKeys(
          accessId: _accessIdCtrl.text.trim(),
          accessSecret: _accessSecretCtrl.text.trim(),
          sampleDeviceId: _sampleDevIdCtrl.text.trim(),
        );
        if (context.mounted) {
          Navigator.pop(context);
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('Harvested and imported $count Wipro bulbs!')),
          );
        }
      } else if (_mode == 4) {
        final count = await home.linkHomeAssistantHub(_urlOrIpCtrl.text, _tokenCtrl.text);
        if (context.mounted) {
          Navigator.pop(context);
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('Imported $count real devices from Hub!')),
          );
        }
      } else if (_mode == 5) {
        final dev = RealSmartDevice(
          id: 'lan_${DateTime.now().millisecondsSinceEpoch}',
          name: _nameCtrl.text.trim().isEmpty ? 'LAN Device' : _nameCtrl.text.trim(),
          room: 'Living Room',
          category: _selectedCategory,
          protocol: _selectedCategory == 'tv'
              ? ConnectionProtocol.onePlusTvLan
              : ConnectionProtocol.tuyaLocalLan,
          credentials: {
            'ip': _urlOrIpCtrl.text.trim(),
            'port': _selectedCategory == 'tv' ? '8008' : '6668',
            'localKey': _localKeyCtrl.text.trim(),
          },
        );
        await home.addDirectLanDevice(dev);
        if (context.mounted) Navigator.pop(context);
      }
    } catch (e) {
      setState(() => _statusMsg = e.toString().replaceAll('Exception: ', ''));
    }
  }

void _showDeviceControlSheet(BuildContext context, RealSmartDevice device) {
  final devId = device.id.toLowerCase();
  final devName = device.name.toLowerCase();
  final cat = device.category.toLowerCase();
  final archetype = UniversalCategoryClassifier.classify(device);
  final isLg = devId.contains('lg') || devName.contains('lg');

  // 1. Camera Cockpit
  if (archetype == UniversalDeviceArchetype.cameraSecurity || cat == 'camera') {
    final home = context.read<RealHomeProvider>();
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      barrierColor: Colors.black.withValues(alpha: 0.65),
      builder: (ctx) => CameraCockpitSheet(
        device: device,
        iot: RealIotController(),
        onStateChanged: () {
          home.notifyUi();
        },
      ),
    );
    return;
  }

  // 2. Laundry Suite (Samsung AI Ecobubble™ Washer vs LG WashTower)
  if (archetype == UniversalDeviceArchetype.laundry ||
      cat == 'washer' ||
      devId.contains('washer') ||
      devName.contains('washer') ||
      devId.contains('washtower')) {
    if (isLg || devId.contains('washtower')) {
      showModalBottomSheet(
        context: context,
        isScrollControlled: true,
        backgroundColor: Colors.transparent,
        barrierColor: Colors.black.withValues(alpha: 0.65),
        builder: (ctx) => LgFutureAppliancesSheet(
          initialDevice: device,
          initialTabIndex: 0,
        ),
      );
      return;
    }
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      barrierColor: Colors.black.withValues(alpha: 0.65),
      builder: (ctx) => SamsungWasherSheet(device: device),
    );
    return;
  }

  // 3. Air Conditioners / Climate Suite (Samsung WindFree™ vs LG DualCool™ / Inverter)
  if (archetype == UniversalDeviceArchetype.climateAc ||
      cat == 'ac' ||
      cat == 'climate' ||
      devId.contains('ac') ||
      devName.contains('ac') ||
      devName.contains('air conditioner')) {
    if (devId.contains('samsung') ||
        devName.contains('samsung') ||
        device.windFreeEnabled ||
        devId.contains('windfree')) {
      showModalBottomSheet(
        context: context,
        isScrollControlled: true,
        backgroundColor: Colors.transparent,
        barrierColor: Colors.black.withValues(alpha: 0.65),
        builder: (ctx) => SamsungFutureDevicesSheet(
          initialDevice: device,
          initialTabIndex: 0,
        ),
      );
      return;
    }
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      barrierColor: Colors.black.withValues(alpha: 0.65),
      builder: (ctx) => LgThinqAcSheet(device: device),
    );
    return;
  }

  // 4. Refrigerators (Samsung Family Hub™ vs LG InstaView™)
  if (archetype == UniversalDeviceArchetype.refrigeration ||
      cat == 'fridge' ||
      cat == 'refrigerator' ||
      devId.contains('fridge') ||
      devName.contains('fridge') ||
      devId.contains('instaview')) {
    if (isLg || devId.contains('instaview')) {
      showModalBottomSheet(
        context: context,
        isScrollControlled: true,
        backgroundColor: Colors.transparent,
        barrierColor: Colors.black.withValues(alpha: 0.65),
        builder: (ctx) => LgFutureAppliancesSheet(
          initialDevice: device,
          initialTabIndex: 1,
        ),
      );
      return;
    }
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      barrierColor: Colors.black.withValues(alpha: 0.65),
      builder: (ctx) => SamsungFutureDevicesSheet(
        initialDevice: device,
        initialTabIndex: 1,
      ),
    );
    return;
  }

  // 5. Air Purifiers / AeroTower (LG PuriCare™ AeroTower)
  if (archetype == UniversalDeviceArchetype.fanAir ||
      cat.contains('purifier') ||
      devId.contains('aerotower') ||
      devName.contains('aerotower') ||
      devId.contains('puricare') ||
      devName.contains('puricare')) {
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      barrierColor: Colors.black.withValues(alpha: 0.65),
      builder: (ctx) => LgFutureAppliancesSheet(
        initialDevice: device,
        initialTabIndex: 2,
      ),
    );
    return;
  }

  // 6. LG Styler® Steam Closet & QuadWash™ Dishwasher
  if (devId.contains('styler') ||
      devName.contains('styler') ||
      devId.contains('dish') ||
      devName.contains('dish')) {
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      barrierColor: Colors.black.withValues(alpha: 0.65),
      builder: (ctx) => LgFutureAppliancesSheet(
        initialDevice: device,
        initialTabIndex: 3,
      ),
    );
    return;
  }

  // 7. Televisions & The Frame Art Mode
  if (archetype == UniversalDeviceArchetype.tvMedia ||
      cat == 'tv' ||
      devId.contains('the_frame') ||
      devName.contains('frame')) {
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      barrierColor: Colors.black.withValues(alpha: 0.65),
      builder: (ctx) => SamsungFutureDevicesSheet(
        initialDevice: device,
        initialTabIndex: 2,
      ),
    );
    return;
  }

  // 8. Fallback: Classic Apple Glass Sheet for generic switches, lights, plugs
  showModalBottomSheet(
    context: context,
    isScrollControlled: true,
    backgroundColor: Colors.transparent,
    barrierColor: Colors.black.withValues(alpha: 0.65),
    builder: (ctx) => _AppleGlassDeviceControlSheet(initialDevice: device),
  );
}

  Widget _buildSmartLightControls(

class _CameraViewfinderPainter extends CustomPainter {
  final bool isPrivacy;
  const _CameraViewfinderPainter({required this.isPrivacy});

  @override
  void paint(Canvas canvas, Size size) {
    if (isPrivacy) return;

    final paint = Paint()
      ..color = Colors.white.withValues(alpha: 0.12)
      ..strokeWidth = 1.0
      ..style = PaintingStyle.stroke;

    // Corner brackets
    const cornerSize = 16.0;
    const pad = 16.0;

    // Top-Left
    canvas.drawLine(const Offset(pad, pad), const Offset(pad + cornerSize, pad), paint);
    canvas.drawLine(const Offset(pad, pad), const Offset(pad, pad + cornerSize), paint);

    // Top-Right
    canvas.drawLine(Offset(size.width - pad, pad), Offset(size.width - pad - cornerSize, pad), paint);
    canvas.drawLine(Offset(size.width - pad, pad), Offset(size.width - pad, pad + cornerSize), paint);

    // Bottom-Left
    canvas.drawLine(Offset(pad, size.height - pad), Offset(pad + cornerSize, size.height - pad), paint);
    canvas.drawLine(Offset(pad, size.height - pad), Offset(pad, size.height - pad - cornerSize), paint);

    // Bottom-Right
    canvas.drawLine(Offset(size.width - pad, size.height - pad), Offset(size.width - pad - cornerSize, size.height - pad), paint);
    canvas.drawLine(Offset(size.width - pad, size.height - pad), Offset(size.width - pad, size.height - pad - cornerSize), paint);
  }

  @override
  bool shouldRepaint(covariant _CameraViewfinderPainter oldDelegate) => oldDelegate.isPrivacy != isPrivacy;
}

class _SplashParticleFieldPainter extends CustomPainter {
  final double progress;
  _SplashParticleFieldPainter({required this.progress});

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2 - 20);
    final maxR = size.longestSide * 0.55;

    for (int i = 0; i < 28; i++) {
      final seed = i * 1.37;
      // Particles spiral inward during 0..0.45, then orbit gently
      final collapse = (1.0 - (progress / 0.48).clamp(0.0, 1.0));
      final dist = 65.0 + (collapse * maxR * ((i % 5 + 1) / 5.0));
      final angle = seed + progress * (i.isEven ? 3.2 : -2.6);
      final pos = Offset(
        center.dx + math.cos(angle) * dist,
        center.dy + math.sin(angle) * dist * 0.75,
      );

      final isWarm = i % 2 == 0;
      final color = isWarm ? const Color(0xFFFFB340) : const Color(0xFF00D4FF);
      final paint = Paint()
        ..color = color.withValues(alpha: (0.18 + (1.0 - collapse) * 0.45).clamp(0.0, 0.75))
        ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 3.0);

      canvas.drawCircle(pos, (i % 3 + 1.8), paint);
    }
  }

  @override
  bool shouldRepaint(covariant _SplashParticleFieldPainter oldDelegate) => true;
}

class _LiquidIrisPortalClipper extends CustomClipper<Path> {
  final double exitProgress;
  _LiquidIrisPortalClipper({required this.exitProgress});

  @override
  Path getClip(Size size) {
    final full = Path()..addRect(Offset.zero & size);
    if (exitProgress <= 0.01) return full;

    final center = Offset(size.width / 2, size.height / 2 - 20);
    final maxRadius = size.longestSide * 1.1;
    final holeRadius = exitProgress * maxRadius;

    final hole = Path();
    const int points = 64;
    for (int i = 0; i <= points; i++) {
      final theta = (i / points) * 2 * math.pi;
      // Organic liquid wave ripple along the expanding portal edge
      final wave = math.sin(theta * 6 + exitProgress * 12) * 14.0 * (1.0 - exitProgress);
      final r = (holeRadius + wave).clamp(0.0, maxRadius * 1.2);
      final pt = Offset(center.dx + math.cos(theta) * r, center.dy + math.sin(theta) * r);
      if (i == 0) {
        hole.moveTo(pt.dx, pt.dy);
      } else {
        hole.lineTo(pt.dx, pt.dy);
      }
    }
    hole.close();

    return Path.combine(PathOperation.difference, full, hole);
  }

  @override
  bool shouldReclip(covariant _LiquidIrisPortalClipper oldClipper) =>
      oldClipper.exitProgress != exitProgress;
}

/// Renders all 12 procedural animations using GPU Canvas primitives
class _UniversalArchetypePainter extends CustomPainter {
  final UniversalDeviceArchetype archetype;
  final double phase;
  final double level;
  final Color tint;

  _UniversalArchetypePainter({
    required this.archetype,
    required this.phase,
    required this.level,
    required this.tint,
  });

  @override
  void paint(Canvas canvas, Size size) {
    switch (archetype) {
      case UniversalDeviceArchetype.lighting:
        _paintLightingBloom(canvas, size);
        break;
      case UniversalDeviceArchetype.climateAc:
        _paintClimateStreamlines(canvas, size);
        break;
      case UniversalDeviceArchetype.fanAir:
        _paintFanVortexRings(canvas, size);
        break;
      case UniversalDeviceArchetype.laundry:
        _paintLaundryWavesAndBubbles(canvas, size);
        break;
      case UniversalDeviceArchetype.refrigeration:
        _paintFridgeSnowflakes(canvas, size);
        break;
      case UniversalDeviceArchetype.tvMedia:
        _paintTvAmbilight(canvas, size);
        break;
      case UniversalDeviceArchetype.speakerAudio:
        _paintSpeakerEqualizerBars(canvas, size);
        break;
      case UniversalDeviceArchetype.robotVacuum:
        _paintVacuumLidarRadar(canvas, size);
        break;
      case UniversalDeviceArchetype.cameraSecurity:
        _paintCameraScanReticle(canvas, size);
        break;
      case UniversalDeviceArchetype.lockAccess:
        _paintLockSecurityShield(canvas, size);
        break;
      case UniversalDeviceArchetype.curtainShade:
        _paintCurtainSunbeams(canvas, size);
        break;
      case UniversalDeviceArchetype.plugEnergy:
        _paintElectricCurrentPulses(canvas, size);
        break;
    }
  }

  void _paintLightingBloom(Canvas canvas, Size size) {
    final radius = size.width * (0.48 + 0.08 * math.sin(phase));
    final paint = Paint()
      ..shader = RadialGradient(
        colors: [
          tint.withValues(alpha: 0.42 * (level / 100.0).clamp(0.25, 1.0)),
          tint.withValues(alpha: 0.0),
        ],
      ).createShader(Rect.fromCircle(center: const Offset(34, 34), radius: radius));
    canvas.drawCircle(const Offset(34, 34), radius, paint);
  }

  void _paintClimateStreamlines(Canvas canvas, Size size) {
    final p = Paint()
      ..color = tint.withValues(alpha: 0.26)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.8;
    for (int i = 0; i < 3; i++) {
      final path = Path();
      final y0 = size.height * (0.32 + i * 0.20);
      for (double x = 0; x <= size.width; x += 6) {
        final y = y0 + math.sin((x / size.width * 2 * math.pi) - phase + i) * 6;
        if (x == 0) {
          path.moveTo(x, y);
        } else {
          path.lineTo(x, y);
        }
      }
      canvas.drawPath(path, p);
    }
  }

  void _paintFanVortexRings(Canvas canvas, Size size) {
    final center = Offset(size.width * 0.72, size.height * 0.42);
    final p = Paint()
      ..color = tint.withValues(alpha: 0.24)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.6;
    for (int i = 0; i < 4; i++) {
      final startAngle = phase + (i * math.pi / 2);
      canvas.drawArc(
        Rect.fromCircle(center: center, radius: 14.0 + i * 9.0),
        startAngle,
        math.pi * 0.65,
        false,
        p,
      );
    }
  }

  void _paintLaundryWavesAndBubbles(Canvas canvas, Size size) {
    final path = Path()..moveTo(0, size.height);
    for (double x = 0; x <= size.width; x += 5) {
      final y = size.height * 0.68 +
          math.sin((x / size.width * 2 * math.pi) + phase) * 7 +
          math.cos((x / size.width * 4 * math.pi) - phase) * 4;
      path.lineTo(x, y);
    }
    path.lineTo(size.width, size.height);
    path.close();
    canvas.drawPath(path, Paint()..color = tint.withValues(alpha: 0.22));

    final bubblePaint = Paint()
      ..color = Colors.white.withValues(alpha: 0.35)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.2;
    for (int i = 0; i < 6; i++) {
      final prog = ((phase / (2 * math.pi)) + i * 0.16) % 1.0;
      final bx = size.width * (0.18 + i * 0.13) + math.sin(phase + i) * 4;
      final by = size.height * (1.0 - prog * 0.82);
      canvas.drawCircle(Offset(bx, by), 2.5 + (i % 3) * 1.5, bubblePaint);
    }
  }

  void _paintFridgeSnowflakes(Canvas canvas, Size size) {
    final dot = Paint()..color = Colors.white.withValues(alpha: 0.45);
    for (int i = 0; i < 8; i++) {
      final prog = ((phase / (2 * math.pi)) + i * 0.125) % 1.0;
      final x = size.width * (0.12 + (i * 0.11)) + math.sin(phase + i) * 4;
      final y = size.height * prog;
      canvas.drawCircle(Offset(x, y), 2.2, dot);
    }
  }

  void _paintTvAmbilight(Canvas canvas, Size size) {
    final rect = Rect.fromLTWH(size.width * 0.15, 0, size.width * 0.7, size.height * 0.45);
    canvas.drawRect(
      rect,
      Paint()
        ..shader = LinearGradient(
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
          colors: [
            tint.withValues(alpha: 0.32 + 0.10 * math.sin(phase)),
            tint.withValues(alpha: 0.0),
          ],
        ).createShader(rect),
    );
  }

  void _paintSpeakerEqualizerBars(Canvas canvas, Size size) {
    final barPaint = Paint()
      ..color = tint.withValues(alpha: 0.35)
      ..strokeCap = StrokeCap.round
      ..strokeWidth = 4.0;
    const barCount = 12;
    final spacing = size.width / (barCount + 2);
    for (int i = 1; i <= barCount; i++) {
      final x = i * spacing;
      final h = 8.0 + 22.0 * ((math.sin(phase * 2 + i * 0.7) + 1) / 2);
      canvas.drawLine(Offset(x, size.height - 6), Offset(x, size.height - 6 - h), barPaint);
    }
  }

  void _paintVacuumLidarRadar(Canvas canvas, Size size) {
    final center = Offset(size.width * 0.75, size.height * 0.40);
    final sweepPaint = Paint()
      ..shader = SweepGradient(
        colors: [tint.withValues(alpha: 0.0), tint.withValues(alpha: 0.40)],
        transform: GradientRotation(phase),
      ).createShader(Rect.fromCircle(center: center, radius: 42));
    canvas.drawCircle(center, 42, sweepPaint);
  }

  void _paintCameraScanReticle(Canvas canvas, Size size) {
    final scanX = size.width * (0.5 + 0.4 * math.sin(phase));
    canvas.drawLine(
      Offset(scanX, 0),
      Offset(scanX, size.height),
      Paint()
        ..color = tint.withValues(alpha: 0.30)
        ..strokeWidth = 1.5,
    );
  }

  void _paintLockSecurityShield(Canvas canvas, Size size) {
    const center = Offset(34, 34);
    for (int i = 1; i <= 2; i++) {
      final prog = ((phase / (2 * math.pi)) + i * 0.5) % 1.0;
      canvas.drawCircle(
        center,
        14 + prog * 36,
        Paint()
          ..color = tint.withValues(alpha: (1.0 - prog) * 0.35)
          ..style = PaintingStyle.stroke
          ..strokeWidth = 1.5,
      );
    }
  }

  void _paintCurtainSunbeams(Canvas canvas, Size size) {
    final openRatio = (level / 100.0).clamp(0.1, 1.0);
    const slatCount = 5;
    final p = Paint()
      ..color = tint.withValues(alpha: 0.22)
      ..strokeWidth = 2.0;
    for (int i = 1; i <= slatCount; i++) {
      final y = (size.height * 0.65 * openRatio) * (i / slatCount);
      canvas.drawLine(Offset(0, y), Offset(size.width, y), p);
    }
  }

  void _paintElectricCurrentPulses(Canvas canvas, Size size) {
    final path = Path();
    final midY = size.height * 0.52;
    for (double x = 0; x <= size.width; x += 8) {
      final spike = math.sin((x / 18.0) - phase * 2) * 8;
      if (x == 0) {
        path.moveTo(x, midY + spike);
      } else {
        path.lineTo(x, midY + spike);
      }
    }
    canvas.drawPath(
      path,
      Paint()
        ..color = tint.withValues(alpha: 0.28)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1.8,
    );
  }

  @override
  bool shouldRepaint(covariant _UniversalArchetypePainter oldDelegate) =>
      oldDelegate.phase != phase || oldDelegate.level != level;
}

    return luminance > 0.45 ? const Color(0xFF101726) : Colors.white;
  }
